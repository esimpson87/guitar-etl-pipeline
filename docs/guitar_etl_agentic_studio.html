C# .NET 8 Semantic Kernel Plugin (GuitarMarketPlugin.cs)

          Microsoft.SemanticKernel 1.x
        

        
using System.ComponentModel;
using System.Net.Http.Json;
using Microsoft.SemanticKernel;

namespace GuitarEtlPipeline.Agents.Plugins;

public sealed class GuitarMarketPlugin(HttpClient httpClient)
{
    [KernelFunction("extract_reverb_listings")]
    [Description("Fetches active and sold guitar listings from the Reverb Developer Hub API and normalizes pricing.")]
    public async Task<string> ExtractReverbListingsAsync(
        [Description("Canonical search query, e.g., 'Fender American Professional II Stratocaster'")] string query,
        [Description("Maximum number of listings to ingest")] int limit = 25,
        CancellationToken ct = default)
    {
        using var request = new HttpRequestMessage(
            HttpMethod.Get,
            $"https://api.reverb.com/api/listings/all?query={Uri.EscapeDataString(query)}&per_page={limit}");
        request.Headers.Add("Accept-Version", "3.0");

        var response = await httpClient.SendAsync(request, ct);
        response.EnsureSuccessStatusCode();
        return await response.Content.ReadAsStringAsync(ct);
    }

    [KernelFunction("query_star_schema_valuation")]
    [Description("Executes a parameterized read-only query against mart_instrument_valuation for comp analysis.")]
    public async Task<InstrumentValuationDto> QueryStarSchemaValuationAsync(
        [Description("Canonical instrument model name")] string canonicalModel,
        CancellationToken ct = default)
    {
        // Connects to BigQuery / Snowflake / Azure SQL Star Schema mart
        return await StarSchemaRepository.GetValuationByModelAsync(canonicalModel, ct);
    }
}

      


      

        

          
dbt Star Schema Mart Model (models/marts/mart_instrument_valuation.sql)

          dbt Core • BigQuery / Snowflake
        

        
{{ config(materialized='table', tags=['star_schema', 'market_intelligence']) }}

WITH sold_comps AS (
    SELECT
        f.listing_id,
        f.instrument_key,
        f.price_usd,
        f.landed_cost_usd,
        f.listed_date
    FROM {{ ref('fact_market_listings') }} f
    WHERE f.listing_type = 'SOLD_COMP'
      AND f.price_usd > 0
)
SELECT
    i.instrument_key,
    i.canonical_model,
    i.brand,
    i.pickup_config,
    i.country_of_origin,
    i.scale_length_in,
    COUNT(s.listing_id) AS sold_comp_count,
    ROUND(AVG(s.price_usd), 2) AS avg_sold_price_usd,
    ROUND(MIN(s.price_usd), 2) AS min_sold_price_usd,
    ROUND(MAX(s.price_usd), 2) AS max_sold_price_usd,
    ROUND(AVG(s.landed_cost_usd), 2) AS avg_landed_sold_usd
FROM {{ ref('dim_instrument') }} i
INNER JOIN sold_comps s
    ON i.instrument_key = s.instrument_key
GROUP BY
    i.instrument_key,
    i.canonical_model,
    i.brand,
    i.pickup_config,
    i.country_of_origin,
    i.scale_length_in
